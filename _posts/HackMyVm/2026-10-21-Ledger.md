le code de notre binaire qui a la permission `SETUID`
```bash
www-data@ledger:~/simpleledger$ cat /opt/ledger-archiver/ledger-archiver.c 
/*
 * ledger-archiver - archive (copy) a ledger file into the protected backup store.
 *
 * The backup storage lives under a root-owned directory, so this helper is
 * installed setuid root in order to read the source and write the destination
 * with root privileges.
 *
 * Usage: ledger-archiver <source-file> <destination-file>
 *
 * Build: gcc -o ledger-archiver ledger-archiver.c
 *        chown root:root ledger-archiver
 *        chmod 4755 ledger-archiver
 */
#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>

int main(int argc, char **argv) {
    if (argc != 3) {
        fprintf(stderr, "Usage: %s <source-file> <destination-file>\n", argv[0]);
        fprintf(stderr, "Archive a ledger file into the protected backup store.\n");
        return 1;
    }

    int fd_in = open(argv[1], O_RDONLY);
    if (fd_in < 0) {
        perror("open source");
        return 1;
    }

    int fd_out = open(argv[2], O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (fd_out < 0) {
        perror("open destination");
        close(fd_in);
        return 1;
    }

    char buf[4096];
    ssize_t n;
    while ((n = read(fd_in, buf, sizeof(buf))) > 0) {
        if (write(fd_out, buf, n) != n) {
            perror("write");
            close(fd_in);
            close(fd_out);
            return 1;
        }
    }

    close(fd_in);
    close(fd_out);
    printf("Archived %s -> %s\n", argv[1], argv[2]);
    return 0;
}
```

c'est code `C` du binaire permet de lire un fichier qui est fournit comme entre et de rédiger la sortie du même fichier avec une destination fournis  et accorde aussi la permission `0644` au fichier de sortie
